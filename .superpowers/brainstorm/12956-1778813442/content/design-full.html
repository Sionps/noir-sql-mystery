-- Who accessed floor 7 after 10pm?
SELECT p.name, h.event, h.timestamp
FROM hotel_log h
JOIN persons p ON h.person_id = p.id
WHERE h.floor = 7
    AND h.timestamp > '22:00'
ORDER BY h.timestamp;
